cmake_minimum_required(VERSION 3.22.1)
project(smdidentity LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "armeabi-v7a")
    add_library(shadowhook SHARED IMPORTED)
    set_target_properties(shadowhook PROPERTIES IMPORTED_LOCATION "${SMD_SHADOWHOOK_DIR}/jni/${ANDROID_ABI}/libshadowhook.so")
    add_library(smdidentity SHARED identity.cpp)
    target_include_directories(smdidentity PRIVATE "${SMD_SHADOWHOOK_DIR}/prefab/modules/shadowhook/include")
    target_link_libraries(smdidentity shadowhook android log dl)
else()
    add_library(smdidentity SHARED unsupported.cpp)
endif()
target_compile_options(smdidentity PRIVATE -Wall -Wextra -fvisibility=hidden)
